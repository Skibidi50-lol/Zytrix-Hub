local function saveOriginal(part)

        part.Size = originalData[part].Size
        part.Transparency = originalData[part].Transparency
        part.CanCollide = originalData[part].CanCollide
    end
end

local espCache = {};
local function createEsp(player)
   local drawings = {};
   
   drawings.box = newDrawing("Square");
   drawings.box.Thickness = 1;
   drawings.box.Filled = false;
   drawings.box.Color = settings.defaultcolor;
   drawings.box.Visible = false;
   drawings.box.ZIndex = 2;

   drawings.boxoutline = newDrawing("Square");
   drawings.boxoutline.Thickness = 3;
   drawings.boxoutline.Filled = false;
   drawings.boxoutline.Color = newColor3();
   drawings.boxoutline.Visible = false;
   drawings.boxoutline.ZIndex = 1;

   espCache[player] = drawings;
end

local function removeEsp(player)
   if rawget(espCache, player) then
       for _, drawing in next, espCache[player] do
           drawing:Remove();
       end
       espCache[player] = nil;
   end
end

local function updateEsp(player, esp)
   local character = player and player.Character;
   if character then
       local cframe = character:GetModelCFrame();
       local position, visible, depth = wtvp(cframe.Position);

       if not settings.enabled then
           esp.box.Visible = false;
           esp.boxoutline.Visible = false;
           return;
       end

       esp.box.Visible = visible;
       esp.boxoutline.Visible = visible;

       if cframe and visible then
           local scaleFactor = 1 / (depth * tan(rad(camera.FieldOfView / 2)) * 2) * 1000;
           local width, height = round(4 * scaleFactor, 5 * scaleFactor);
           local x, y = round(position.X, position.Y);

           esp.box.Size = newVector2(width, height);
           esp.box.Position = newVector2(round(x - width / 2, y - height / 2));
           esp.box.Color = settings.teamcolor and player.TeamColor.Color or settings.defaultcolor;

           esp.boxoutline.Size = esp.box.Size;
           esp.boxoutline.Position = esp.box.Position;
       end
   else
       esp.box.Visible = false;
       esp.boxoutline.Visible = false;
   end
end

local function IsVisible(targetPart)
    local origin = Camera.CFrame.Position
    local direction = (targetPart.Position - origin)

    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {LocalPlayer.Character}
    params.FilterType = Enum.RaycastFilterType.Blacklist

    local result = workspace:Raycast(origin, direction, params)

    if result then
        return result.Instance:IsDescendantOf(targetPart.Parent)
    end

    return false
end

local function GetClosestPlayer()
    local closest = nil
    local shortest = FOV

    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("Head") and plr.Character:FindFirstChild("HumanoidRootPart") then
            
            local root = plr.Character.HumanoidRootPart
            local head = plr.Character.Head

            local distance3D = (root.Position - Camera.CFrame.Position).Magnitude
            if distance3D <= 1000 then

                local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local dist2D = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude

                    if dist2D < shortest and IsVisible(head) then
                        shortest = dist2D
                        closest = plr
                    end
                end

            end
        end
    end

    return closest
end

local function escapeFromBombHolder(bombPosition)

	if not dodgeEnabled then return end

	local now = tick()

	if now - lastEscapeTime < COOLDOWN then
		return
	end

	local char = player.Character
	if not char then return end

	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then return end

	local myPosition = root.Position

	local awayVector = (myPosition - bombPosition)

	local direction

	if awayVector.Magnitude > 0.1 then
		direction = awayVector.Unit
	else
		direction = Vector3.new(
			math.random()-0.5,
			0,
			math.random()-0.5
		).Unit
	end

	local newPosition = myPosition + (direction * ESCAPE_DISTANCE)

	root.CFrame = CFrame.new(
		Vector3.new(newPosition.X,myPosition.Y,newPosition.Z)
	)

	lastEscapeTime = now
end

local function startAutoDodge()

	if heartbeatConnection then
		heartbeatConnection:Disconnect()
	end

	heartbeatConnection = RunService.Heartbeat:Connect(function()

		if not dodgeEnabled then
			return
		end

		local char = player.Character
		if not char then return end

		local myRoot = char:FindFirstChild("HumanoidRootPart")
		if not myRoot then return end

		local myPosition = myRoot.Position

		for _,v in pairs(Players:GetPlayers()) do

			if v == player then
				continue
			end

			if not v.Character then
				continue
			end

			local hasBomb = false

			for _,tool in pairs(v.Character:GetChildren()) do

				if tool:IsA("Tool") then

					local name = tool.Name:lower()

					if name:find("bomb")
					or name:find("grenade")
					or name:find("explosive") then

						hasBomb = true
						break
					end
				end
			end

			if hasBomb then

				local bombRoot = v.Character:FindFirstChild("HumanoidRootPart")

				if bombRoot then

					local distance = (
						myPosition - bombRoot.Position
					).Magnitude

					if distance <= DETECTION_RADIUS then
						escapeFromBombHolder(bombRoot.Position)
						break
					end
				end
			end
		end
	end)
end

task.spawn(function()

	while true do

		task.wait(0.1)

		if not enabled then
			currentTarget = nil
			continue
		end

		local char = player.Character
		if not char then continue end

		local root = char:FindFirstChild("HumanoidRootPart")
		if not root then continue end

		local tool = char:FindFirstChildOfClass("Tool")
		if not tool then continue end

		local myPos = root.Position

		local bestTarget = nil
		local bestDist = math.huge

		for _,v in pairs(Players:GetPlayers()) do

			if v == player then continue end
			if not v.Character then continue end

			local tr = v.Character:FindFirstChild("HumanoidRootPart")
			local hum = v.Character:FindFirstChild("Humanoid")

			if not tr or not hum then continue end
			if hum.Health <= 0 then continue end

			local d = (myPos - tr.Position).Magnitude

			if d < bestDist then
				bestDist = d
				bestTarget = v
			end
		end

		currentTarget = bestTarget
	end
end)

		local targetRoot = currentTarget.Character:FindFirstChild("HumanoidRootPart")
		local targetHum = currentTarget.Character:FindFirstChild("Humanoid")

		if targetRoot and targetHum and targetHum.Health > 0 then

			local targetPos = targetRoot.Position

				hum:MoveTo(targetPos)

				if (root.Position - targetPos).Magnitude < 2 then
					hum:MoveTo(targetPos + targetRoot.CFrame.LookVector * 2)

			elseif auraMode == "Legit" then

				local distance = (root.Position - targetPos).Magnitude

				if distance > 6 then

						targetPos - (
							(targetPos - root.Position).Unit * 4
						)
					)

				elseif distance <= 6 and distance >= 3 then

					hum:MoveTo(
						targetPos - (
							(targetPos - root.Position).Unit * 2.5
						)
					)
				end
			end
		end
	end
end)

-- PROX PROMPT
local function triggerPrompt(prompt)
	if fireproximityprompt then
		fireproximityprompt(prompt)
	else
		prompt:InputHoldBegin()
		task.wait(prompt.HoldDuration or 0.1)
		prompt:InputHoldEnd()

local function hasPrompt(barrel)
	for _, v in ipairs(barrel:GetDescendants()) do
		if v:IsA("ProximityPrompt") then
			return true
		end
	end
	return false
end

-- GET BARRELS
local function getBarrels()
	local barrels = {}

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") and obj.Name:lower():find("barrel") and hasPrompt(obj) then
			table.insert(barrels, obj)
		end
	end

	-- randomize
	for i = #barrels, 2, -1 do
		local j = math.random(i)
		barrels[i], barrels[j] = barrels[j], barrels[i]
	end

	return barrels
end

local function removeHighlight(model)
	if Highlights[model] then
		Highlights[model]:Destroy()
		Highlights[model] = nil
	end
end

local function createHighlight(model)
	if not _G.BarrelHighlightEnabled then
		removeHighlight(model)
		return
	end

	if Highlights[model] then
		return
	end

	local h = Instance.new("Highlight")
	h.Name = "AutoHighlight"
	h.FillColor = Color3.fromRGB(255, 170, 0)
	h.OutlineColor = Color3.fromRGB(255, 255, 255)
	h.FillTransparency = 0.4
	h.OutlineTransparency = 0
	h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	h.Parent = model

	Highlights[model] = h
end

local function getClosestNPC()
	local closest = nil
	local shortestDist = math.huge

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") and obj ~= player.Character then

			local hum = obj:FindFirstChildWhichIsA("Humanoid")
			local head = obj:FindFirstChild("Head")

			if hum and head and hum.Health > 0 then

				local isPlayer = false

				for _, plr in ipairs(Players:GetPlayers()) do
					if plr.Character == obj then
						isPlayer = true
						break
					end
				end

				if not isPlayer then
					local dist = (head.Position - camera.CFrame.Position).Magnitude

					if dist < shortestDist then
						shortestDist = dist
						closest = head
					end
				end
			end
		end
	end

	return closest
end

local function updateCamlock()
	if camlock then
		if not camlockLoop then
			camlockLoop = RunService.RenderStepped:Connect(function()
				local target = getClosestNPC()

				if target then
					camera.CFrame = CFrame.new(
						camera.CFrame.Position,
						target.Position
					)
				end
			end)
		end
	else
		if camlockLoop then
			camlockLoop:Disconnect()
			camlockLoop = nil
		end
	end
end

local function teleportToNearestBarrel()
	local _, hrp = getCharacter()

	local closestBarrel = nil
	local shortestDistance = math.huge

	for _, barrel in ipairs(getBarrels()) do
		local part = barrel.PrimaryPart or barrel:FindFirstChildWhichIsA("BasePart")

		if part then
			local distance = (hrp.Position - part.Position).Magnitude

			if distance < shortestDistance then
				shortestDistance = distance
				closestBarrel = part
			end
		end
	end

	if closestBarrel then
		hrp.CFrame = closestBarrel.CFrame * CFrame.new(0, 0, 3)

		StarterGui:SetCore("SendNotification", {
			Title = "Barrel TP",
			Text = "Teleported to nearest barrel",
			Duration = 2
		})
	else
		StarterGui:SetCore("SendNotification", {
			Title = "Barrel TP",
			Text = "No barrels found",
			Duration = 2
		})
	end
end

_G.DisableParticles = false

local function optimizeGame()

	for _, v in ipairs(game:GetDescendants()) do

		-- REMOVE TEXTURES
		if _G.DisableTextures then
			if v:IsA("Texture") or v:IsA("Decal") then
				v.Transparency = 1
			end
		end

		-- DISABLE PARTICLES
		if _G.DisableParticles then
			if v:IsA("ParticleEmitter")
			or v:IsA("Trail")
			or v:IsA("Smoke")
			or v:IsA("Fire")
			or v:IsA("Sparkles") then
				v.Enabled = false
			end
		end

		-- DISABLE EFFECTS
		if _G.DisableEffects then
			if v:IsA("BloomEffect")
			or v:IsA("BlurEffect")
			or v:IsA("SunRaysEffect")
			or v:IsA("ColorCorrectionEffect")
			or v:IsA("DepthOfFieldEffect") then
				v.Enabled = false
			end
		end

		-- LOW QUALITY PARTS
		if _G.LowQualityParts then
			if v:IsA("BasePart") then
				v.Material = Enum.Material.Plastic
				v.Reflectance = 0
			end
		end
	end

	-- REMOVE SHADOWS
	if _G.RemoveShadows then
		Lighting.GlobalShadows = false
		Lighting.FogEnd = 999999
		Lighting.Brightness = 1
	end

	if _G.FPSBoost then
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01

		for _, v in ipairs(workspace:GetDescendants()) do
			if v:IsA("BasePart") then
				v.CastShadow = false
			end
		end
	end
end
